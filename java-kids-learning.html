📝 Class = Blueprint
স্কুলে টুর্নামেন্টের জন্য স্যার একটা খালি ফর্ম বানিয়েছেন। সেই খালি ফর্মটাই হলো Class (নকশা)। 
public class Student {
    String name;      // নামের জায়গা
    int roll;         // রোলের জায়গা
    String className; // শ্রেণির জায়গা
    String section;   // শাখার জায়গা
}

🧒 Object = ভরাট ফর্ম

💻 পুরো কোড + রান
public class Student {
    String name;
    int roll;
    String className;
    String section;

    void showInfo() {
        System.out.println("নাম: " + name);
        System.out.println("রোল: " + roll);
        System.out.println("শ্রেণি: " + className);
        System.out.println("শাখা: " + section);
    }

    public static void main(String[] args) {
        Student rahim = new Student();
        rahim.name = "রহিম";
        rahim.roll = 5;
        rahim.className = "নবম";
        rahim.section = "ক";

        Student karim = new Student();
        karim.name = "করিম";
        karim.roll = 12;
        karim.className = "নবম";
        karim.section = "খ";

        System.out.println("রহিমের ফর্ম:");
        rahim.showInfo();
        System.out.println("করিমের ফর্ম:");
        karim.showInfo();
    }
}
